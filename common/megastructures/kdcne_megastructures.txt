# Both operation stages finish naturally and remain valid launchers. Each has
# one outgoing operation, so repeated use never needs a scripted reverse upgrade.
kdcne_mega_cloud_projector_control = {
    entity = "dyson_sphere_phase_01_entity"
    portrait = "GFX_megastructure_construction_background"
    place_entity_on_planet_plane = no
    entity_offset = { x = -27 y = -27 }
    construction_blocks_and_blocked_by = none
    show_galactic_map_icon = yes
    build_time = 1
    resources = { category = megastructures upkeep = { energy = 100 } }
    upgrade_from = { kdcne_mega_cloud_projector kdcne_mega_cloud_projector_control_repeat }
    upgrade_desc = hide
    potential = { is_ai = no }
    possible = {
        custom_tooltip = {
            fail_text = kdcne_requires_no_projection_window
            from = { kdcne_projection_window_free = yes }
        }
        from = { kdcne_projection_cooldown_ready = yes }
    }
    on_build_complete = {
        fromfrom = {
            set_megastructure_flag = kdcne_projector_ready
            save_event_target_as = kdcne_projection_opening_launcher
            if = {
                limit = { exists = owner }
                owner = {
                    set_timed_country_flag = { flag = kdcne_projection_open_pending days = 2 }
                    country_event = { id = kdcne_cloud.24 days = 1 }
                }
            }
        }
    }
    ai_weight = { factor = 0 }
}
