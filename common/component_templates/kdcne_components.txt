# --- 纳米鱼雷（G 槽，命中施加纳米侵蚀） -----------------------------------------
weapon_component_template = {
	key = "kdcne_nano_torpedo"
	size = torpedo
	entity = "invisible_turret_entity"
	type = missile
	icon = "GFX_wg_gray_g"
	icon_frame = 1
	component_set = "kdcne_nano_torpedo"
	projectile_gfx = "gatebuilder_torpedo"
	tags = { kdcne_nano_torpedo weapon_type_nano weapon_type_explosive }
	ai_tags = { weapon_role_anti_shield }

	damage = { min = 800 max = 1600 }
	windup = { min = 5 max = 20 }
	total_fire_time = 130
	range = 180.0
	accuracy = 1.0
	tracking = 0.5
	power = -150

	shield_penetration = 0.5
	armor_penetration = 0.5
	hull_damage = 1.2

	# 只对当前纳米组件载舰生效，分别由对应科技解锁。
	triggered_ship_modifier = {
		potential = { exists = owner owner = { has_technology = tech_kdcne_lifesteal } }
		component_lifesteal_add = 0.1
	}
	# Assimilation is handled only by prolonged cloud exposure and the player toggle.

	missile_speed = 30
	missile_evasion = 0.10
	missile_shield = 50
	missile_armor = 10
	missile_health = 50
	missile_retarget_range = 100

	resources = {
		category = ship_components
		cost = { alloys = 10 nanites = 5 }
		upkeep = { energy = 5 alloys = 1 }
	}

	# 命中生成固定云区；云区逐日累积侵蚀。
	on_hit = {
		custom_tooltip = "kdcne_nano_corrosion_tooltip"
		hidden_effect = {
			# 混装纳米舰载机时统一走受伤钩子，避免同次鱼雷伤害重复判定。
			if = {
				limit = { NOT = { from = { OR = { has_component = kdcne_nano_aircraft has_component = kdcne_nano_cloud_generator } } } }
				kdcne_spawn_hit_cloud = yes
			}
		}
	}

	prerequisites = { "tech_kdcne_nano_corrosion" }

	ai_weight = { weight = 0 }
}
