# Planet: also used on save load; changes only the rendered entity, never the body/class.
kdcne_refresh_nano_star_model = {
    set_planet_entity = {
        entity = "black_hole_entity"
        picture = "pc_wg_graygoo"
        atmosphere_color = { 0.6 0.6 0.6 }
        atmosphere_intensity = 1.0
        atmosphere_width = 0.5
    }
}
# Planet: artificial worlds are included. Do not leave orbital fleets on removed bodies.
kdcne_remove_stellar_body = {
    if = { limit = { exists = orbital_station } orbital_station = { dismantle = yes } }
    if = { limit = { exists = orbital_defence } destroy_fleet = orbital_defence }
    if = { limit = { is_colony = yes } destroy_colony = yes }
    clear_deposits = yes
    remove_planet = yes
}
