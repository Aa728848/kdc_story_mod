# Natural completion hands off the exact instance. No stage restoration is needed.
country_event = {
    id = kdcne_cloud.24
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        remove_country_flag = kdcne_projection_open_pending
        if = {
            limit = { kdcne_projection_window_free = yes }
            kdcne_projection_clear_selection = yes
            if = {
                limit = { exists = event_target:kdcne_projection_opening_launcher }
                event_target:kdcne_projection_opening_launcher = {
                    if = {
                        limit = {
                            kdcne_projection_stage_valid = yes
                            exists = owner
                            if = { limit = { exists = owner } owner = { is_same_value = root } }
                        }
                        save_global_event_target_as = kdcne_projection_launcher_@root
                    }
                }
            }
            if = {
                limit = { kdcne_projection_launcher_valid = { SOURCE = root } }
                log = "KDCNE projector: completed stage bound; opening control"
                country_event = { id = kdcne_cloud.21 }
            }
            else = { country_event = { id = kdcne_cloud.26 } }
        }
    }
}

# Explicit maintenance for already damaged instances from earlier versions.
# Valid stages are left intact. The replacement must be verified before removal.
country_event = {
    id = kdcne_cloud.25
    title = kdcne_cloud.25.title
    desc = kdcne_cloud.25.desc
    picture = GFX_evt_landing_ship
    is_triggered_only = yes
    option = {
        name = kdcne_repair_projectors
        allow = { kdcne_projection_window_free = yes }
        if = {
            limit = { kdcne_projection_window_free = yes }
            kdcne_projection_clear_selection = yes
            every_owned_megastructure = {
                limit = {
                    has_megastructure_flag = kdcne_projector_ready
                    NOR = {
                        is_megastructure_type = kdcne_mega_cloud_projector
                        is_megastructure_type = kdcne_mega_cloud_projector_control
                        is_megastructure_type = kdcne_mega_cloud_projector_control_repeat
                    }
                    is_upgrading = no
                    exists = planet
                    exists = solar_system
                    exists = owner
                    if = { limit = { exists = owner } owner = { is_same_value = root } }
                }
                root = { remove_country_flag = kdcne_repair_spawned }
                save_event_target_as = kdcne_repair_old
                planet = { save_event_target_as = kdcne_repair_planet }
                solar_system = {
                    spawn_megastructure = {
                        type = kdcne_mega_cloud_projector
                        owner = root
                        planet = event_target:kdcne_repair_planet
                        coords_from = event_target:kdcne_repair_old
                        graphical_culture = root
                        init_effect = {
                            set_megastructure_flag = kdcne_projector_ready
                            save_event_target_as = kdcne_repair_new
                            root = { set_country_flag = kdcne_repair_spawned }
                        }
                    }
                }
                if = {
                    limit = {
                        root = { has_country_flag = kdcne_repair_spawned }
                        exists = event_target:kdcne_repair_new
                        if = {
                            limit = { exists = event_target:kdcne_repair_new }
                            event_target:kdcne_repair_new = {
                                kdcne_projection_stage_valid = yes
                                NOT = { is_same_value = event_target:kdcne_repair_old }
                                exists = owner
                                if = { limit = { exists = owner } owner = { is_same_value = root } }
                                exists = planet
                                if = { limit = { exists = planet } planet = { is_same_value = event_target:kdcne_repair_planet } }
                            }
                        }
                    }
                    remove_megastructure = this
                }
            }
            remove_country_flag = kdcne_repair_spawned
        }
    }
    option = { name = kdcne_close }
}
